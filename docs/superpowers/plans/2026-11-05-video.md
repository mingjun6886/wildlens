> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a signed-in user upload an MP4, MOV or AVI of up to 100 MB and get it tagged (MaxN over one frame per second, at most 60 frames) by the existing `process` function.

**Architecture:** A new pure flat module `services/process/video.py` samples frames with OpenCV and combines per-frame tags. `app.py` routes video suffixes to a video branch that tags frames one at a time under a Lambda time guard and records `type="video"`, `frameCount` and `durationSec`. **Tech Stack:** Python 3.11, OpenCV (`opencv-python-headless==5.0.0.93`), pytest, boto3; framework-free JS tested with `node --test`.

@pytest.fixture
def make_video(tmp_path):
    """Write a small synthetic video and return its path.

    Generated rather than committed, so no binary sample lives in the repository.
    MJPG in an AVI is written by OpenCV itself, without FFmpeg's encoders; mp4v in
    an MP4 goes through FFmpeg. Each frame is a flat grey whose level is its index,
    so a test can tell which frames were kept by reading them back.
    """
    import cv2
    import numpy as np

    def write(name: str, seconds: float, fps: float, size: tuple[int, int] = (64, 48)) -> Path:
        path = tmp_path / name
        fourcc = "MJPG" if path.suffix == ".avi" else "mp4v"
        writer = cv2.VideoWriter(str(path), cv2.VideoWriter_fourcc(*fourcc), fps, size)
        assert writer.isOpened(), f"OpenCV could not open a writer for {name}"
        for index in range(int(seconds * fps)):
            writer.write(np.full((size[1], size[0], 3), index % 256, np.uint8))
        writer.release()
        return path

    return write

```python
"""Turn a video file into still frames, and per-frame tags into one result.

Pure: OpenCV and the filesystem only, so all of it is tested without models or
AWS. app.py decides what to do with the frames.
"""

from __future__ import annotations

import math
from dataclasses import dataclass
from pathlib import Path

import cv2

VIDEO_SUFFIXES = {".mp4", ".mov", ".avi"}

# One frame per second, the first minute. At about 7 s of inference per frame
# warm, 60 frames is about 7 minutes, inside the 900 s timeout with room for a
# 41 s cold start and the thumbnail.
MAX_FRAMES = 60

# Some AVI and MOV files report no frame rate. 30 is the most common camera rate,
# and a wrong guess only changes which second each kept frame comes from.
FALLBACK_FPS = 30.0

# Frames are inputs to the detector, not thumbnails, so quality stays high.
FRAME_JPEG_QUALITY = 95


class UnreadableVideo(Exception):
    """The file is not a video OpenCV can read any frame from."""


@dataclass(frozen=True)
class SampledVideo:
    frames: list[Path]
    duration_sec: float | None
    fps_assumed: bool


def effective_fps(reported: float) -> tuple[float, bool]:
    """The frame rate to sample by, and whether it was assumed."""
    if not math.isfinite(reported) or reported <= 0:
        return FALLBACK_FPS, True
    return reported, False


def sample_frames(path: Path, out_dir: Path, max_frames: int = MAX_FRAMES) -> SampledVideo:
    """Write one JPEG per second of video, at most `max_frames`, in time order.

    Reads sequentially rather than seeking: seeking lands on the wrong frame in
    some AVI and MOV files. grab() advances without converting the frame to an
    image, so the skipped frames cost a decode but not a copy.

    A read that fails part-way through ends sampling and keeps what was read. A
    file that yields no frame at all raises UnreadableVideo.
    """
    capture = cv2.VideoCapture(str(path))
    try:
        if not capture.isOpened():
            raise UnreadableVideo("OpenCV could not open the file")

        fps, assumed = effective_fps(capture.get(cv2.CAP_PROP_FPS))
        step = max(1, round(fps))

        # A renamed JPEG reports a frame count of about -9.2e18, so the count is
        # trusted only when it is finite and positive and the rate was reported.
        total = capture.get(cv2.CAP_PROP_FRAME_COUNT)
        duration = (
            round(total / fps, 1) if not assumed and math.isfinite(total) and total > 0 else None
        )

        out_dir.mkdir(parents=True, exist_ok=True)
        frames: list[Path] = []
        index = 0
        while len(frames) < max_frames and capture.grab():
            if index % step == 0:
                ok, image = capture.retrieve()
                if not ok:
                    break
                destination = out_dir / f"frame-{len(frames):03d}.jpg"
                if not cv2.imwrite(
                    str(destination), image, [cv2.IMWRITE_JPEG_QUALITY, FRAME_JPEG_QUALITY]
                ):
                    # Disk full or similar: not a property of the video, so it is
                    # raised as an ordinary error and SQS retries it.
                    raise OSError(f"could not write {destination}")
                frames.append(destination)
            index += 1
    finally:
        capture.release()

    if not frames:
        raise UnreadableVideo("no frame could be read")
    return SampledVideo(frames=frames, duration_sec=duration, fps_assumed=assumed)


def max_n(per_frame: list[dict[str, int]]) -> dict[str, int]:
    """For each species, the largest count seen in any single frame.

    Summing would count one animal standing still once per second. MaxN is the
    standard camera-trap measure and never exceeds what was visible at once.
    """
    result: dict[str, int] = {}
    for tags in per_frame:
        for name, count in tags.items():
            result[name] = max(result.get(name, 0), count)
    return result


def busiest_frame(per_frame: list[dict[str, int]]) -> int:
    """Index of the frame with the most animals; the earliest wins a tie.

    The thumbnail comes from here rather than the first frame, because the first
    second of a trap clip is often empty: the animal is what triggered it.
    """
    totals = [sum(tags.values()) for tags in per_frame]
    return totals.index(max(totals))
```

def store_thumbnail(source: Path, thumb_key: str) -> None:
    """Make a JPEG thumbnail of `source` and upload it under `thumb_key`."""
    thumb_path = TMP / f"thumb-{uuid.uuid4().hex}.jpg"
    try:
        with span("thumbnail"):
            make_thumbnail(source, thumb_path)
        # ContentType is set explicitly because S3 does not infer it. Without
        # it the object is served as binary/octet-stream, which a browser may
        # still render by sniffing but will offer to download rather than
        # display, and which gives the wrong type to anything that trusts the
        # header. make_thumbnail always writes JPEG, whatever came in.
        s3.upload_file(
            str(thumb_path),
            os.environ["THUMB_BUCKET"],
            thumb_key,
            ExtraArgs={"ContentType": "image/jpeg"},
        )
    finally:
        thumb_path.unlink(missing_ok=True)


def tag_still(local_path: Path, file_id: str, suffix: str, correlation_id: str) -> Tagged:
    """Verify, tag and thumbnail one image."""
    # (move the existing readability comment block and the Image.open/verify try
    # from process_object here, unchanged)

        with Image.open(local_path) as probe:

    thumb_key = f"{file_id}{suffix}"
    store_thumbnail(local_path, thumb_key)
    return Tagged("image", tags, model_load_ms, inference_ms, thumb_key)


def tag_video(
    local_path: Path, file_id: str, correlation_id: str, remaining_ms: Callable[[], int]
) -> Tagged:
    """Sample, tag frame by frame under the time guard, and combine with MaxN.

    Frames are tagged one at a time with tag_image rather than in one tag_images
    call, because tag_images detects every frame before classifying any, and the
    time guard could not interrupt it.
    """
    frame_dir = TMP / f"frames-{uuid.uuid4().hex}"
    try:
        try:
            sampled = sample_frames(local_path, frame_dir)
        except UnreadableVideo as error:
            raise PermanentFailure(f"not a readable video: {error}") from error
        if sampled.fps_assumed:
            log_event("video frame rate assumed", correlation_id, fileId=file_id)

        (detector, species_model, classes, label_map), model_load_ms = load_models()

        per_frame: list[dict[str, int]] = []
        started = time.time()
        for frame in sampled.frames:
            if remaining_ms() < VIDEO_TIME_MARGIN_MS:
                log_event(
                    "video truncated",
                    correlation_id,
                    fileId=file_id,
                    framesTagged=len(per_frame),
                    framesSampled=len(sampled.frames),
                )
                break
            with span("tagging"):
                per_frame.append(
                    tag_image(str(frame), detector, species_model, classes, label_map)
                )
        tagging_ms = int((time.time() - started) * 1000)

        if not per_frame:
            # A retry would meet the same cold start and the same clock.
            raise PermanentFailure("ran out of time before tagging any frame")

        thumb_key = f"{file_id}.jpg"
        store_thumbnail(sampled.frames[busiest_frame(per_frame)], thumb_key)

        extra = {"frameCount": Decimal(len(per_frame))}
        if sampled.duration_sec is not None:
            extra["durationSec"] = Decimal(str(sampled.duration_sec))

        # The mean per frame, so InferenceDuration keeps meaning "time to tag one
        # image" and the dashboard's percentiles stay comparable with images.
        return Tagged(
            "video", max_n(per_frame), model_load_ms, tagging_ms // len(per_frame), thumb_key, extra
        )
    finally:
        shutil.rmtree(frame_dir, ignore_errors=True)
```

 * Upload limits and polling by kind of file. Mirrors services/api/upload.py: the
 * API is the authority, and this only spares the browser from hashing a file the
 * API would refuse.
 */

export const MAX_IMAGE_BYTES = 25 * 1024 * 1024;
export const MAX_VIDEO_BYTES = 100 * 1024 * 1024;
export const VIDEO_EXTENSIONS = ["mp4", "mov", "avi"];

export function isVideo(name) {
  const match = /\.([a-z0-9]+)$/i.exec(name);
  return Boolean(match) && VIDEO_EXTENSIONS.includes(match[1].toLowerCase());
}

export function checkSize(file) {
  const video = isVideo(file.name);
  const limit = video ? MAX_VIDEO_BYTES : MAX_IMAGE_BYTES;
  if (file.size > limit) {
    throw new Error(
      `${file.name} is ${(file.size / 1024 / 1024).toFixed(1)} MB; ` +
        `${video ? "a video" : "an image"} may be up to ${limit / 1024 / 1024} MB`,
    );
  }
}

/**
 * How often and how long to poll. An image settles in seconds. A video takes
 * about 7 s a frame for up to 60 frames, under the function's 900 s timeout, plus
 * a cold start and any queue: 16 minutes covers the worst legitimate case, and a
 * 10 s interval keeps that to under a hundred requests.
 */
export function pollPlan(name) {
  return isVideo(name)
    ? { intervalMs: 10_000, timeoutMs: 16 * 60_000 }
    : { intervalMs: 2000, timeoutMs: 120_000 };
}
```
